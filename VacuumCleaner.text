
A = "Dirty"
B = "Dirty"
location = "A"

print("Initial State:")
print("A =", A)
print("B =", B)
print("Vacuum Cleaner at:", location)

print("\nAgent Actions:")

for i in range(6):

    if location == "A":
        if A == "Dirty":
            print("A is Dirty -> Suck")
            A = "Clean"
        else:
            print("A is Clean -> Move Right")
            location = "B"

    else:
        if B == "Dirty":
            print("B is Dirty -> Suck")
            B = "Clean"
        else:
            print("B is Clean -> Move Left")
            location = "A"

print("A =", A)
print("B =", B)     
print("Vacuum Cleaner at:", location)
