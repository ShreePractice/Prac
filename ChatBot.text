print("Chatbot: Hello! Welcome to Simple Chatbot.")
print("Chatbot: You can ask me about my name, age, hobbies, or time.")
print("Chatbot: Type 'bye' to exit.\n")
while True:
    user = input("You: ").lower()
    if "hello" in user or "hi" in user:
        print("Chatbot: Hello! How can I help you?")
    elif "name" in user:
        print("Chatbot: I am a simple Python chatbot.")
    elif "how are you" in user:
        print("Chatbot: I am fine! Thank you.")
    elif "age" in user:
        print("Chatbot: I am just a computer program.")
    elif "hobby" in user:
        print("Chatbot: I like talking with people.")
    elif "time" in user:
        print("Chatbot: Sorry, I cannot tell the current time.")
    elif "help" in user:
        print("Chatbot: You can ask me about my name, age, or hobbies.")
    elif "bye" in user or "exit" in user:
        print("Chatbot: Goodbye! Have a nice day.")
        break
    else:
        print("Chatbot: Sorry, I don't understand that.")
